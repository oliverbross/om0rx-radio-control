│                                                                           │
│         qso = {                                                          │

│         }                                                                │
│                                                                           │
│         # Parse ADIF fields: <fieldname:length>value                   │
│         for match in re.finditer(r'<(\w+):(\d+)>([^<]*)', record):    │
│             field_name = match.group(1).lower()                        │
│             field_value = match.group(3)                               │
│                                                                           │
│             # Map to dictionary keys                                    │
│             if field_name == 'call':                                    │
│                 qso['call'] = field_value.upper()                      │
│             # ... (handle other fields)                                 │
│                                                                           │
│         if qso['call']:                                                 │
│             qsos.append(qso)                                            │
│             if len(qsos) >= limit:                                      │
│                 break                                                    │
│                                                                           │
│     return qsos                                                          │
